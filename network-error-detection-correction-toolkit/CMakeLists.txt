cmake_minimum_required(VERSION 3.10)
project(EDC_Toolkit CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(edc_codes STATIC src/codes.cpp src/errors.cpp)
target_include_directories(edc_codes PUBLIC src)
add_library(edc_net STATIC src/sha256.cpp src/link.cpp)
target_link_libraries(edc_net PUBLIC edc_codes Threads::Threads)
if(WIN32)
    target_link_libraries(edc_net PUBLIC ws2_32)
endif()
add_executable(edc_toolkit src/toolkit_main.cpp)
target_link_libraries(edc_toolkit edc_codes)
foreach(p sender receiver channel bench)
    add_executable(edc_${p} src/${p}_main.cpp)
    target_link_libraries(edc_${p} edc_net)
endforeach()
add_executable(edc_tests tests/test_main.cpp)
target_link_libraries(edc_tests edc_net)
